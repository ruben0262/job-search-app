# Tooling that runs against the whole tree — not shipped in any image.
#
# Per-service runtime dependencies stay in services/<name>/requirements.txt (CODE_STRUCTURE.md
# §5): docgen is the only image with Chrome, and keeping that split is what makes api and notifier
# fast to rebuild.
#
#   pip install -r requirements-dev.txt

import-linter>=2.0        # enforces the boundary contracts in pyproject.toml — `lint-imports`
pytest>=8.0
ruff>=0.8

# libs/db is imported by nearly every service, so its driver is needed to run the tests. Services still
# pin their own runtime copies; these are the versions local development resolves against. Capped at
# the major version: a driver release must not change what the gates say about unchanged code.
pymongo>=4.19,<5
pydantic>=2.10        # libs/contracts: one model behind server validation and the form
# services/harvester's fetch layer. Declared here because tests import it; the service pins its own
# copy in services/harvester/requirements.txt. It went undeclared for one commit and CI caught it -
# locally it was present only as a transitive dependency of the embedder's test deps.
httpx>=0.28
# Several adapters parse RSS/XML from third-party job boards. Stdlib ElementTree is documented as
# vulnerable to entity-expansion bombs, and the harvester shares a box with Postgres.
defusedxml>=0.7
# libs/secrets: AES-256-GCM for the KEK/DEK envelope. Every service that can hold a user credential
# needs it at runtime and pins its own copy; it is here because libs/secrets and its T4 tests are
# imported by the root suite.
cryptography>=44.0
# libs/auth: argon2id with FlowSolutions' production parameters (AUTH_REUSE.md §3). The
# primitive comes from a maintained library rather than being hand-rolled - that is the whole
# argument for porting the design instead of the Go code.
argon2-cffi>=23.1
# services/api. FastAPI and PyJWT are here because the auth endpoint tests exercise the real
# app rather than a stand-in - a cookie policy asserted against a mock proves nothing.
fastapi>=0.115
pydantic[email]>=2.10
PyJWT>=2.9
# libs/documents: the upload path extracts text from CVs and references. The tests build real
# .docx and .pdf bytes rather than mocking a parser - a parser mock proves nothing about
# whether a real CV reads.
pypdf>=5.1
python-docx>=1.1
python-multipart>=0.0.20

# libs/corpus's posting fetcher and language filter. Both fail open by design — a page it cannot
# parse is "no signal", a language it cannot detect is "keep the job" — which is right in
# production and wrong in CI: without these installed the tests did not skip, they asserted the
# fallback and failed. Declared here so the root suite exercises the real parsers; the notifier
# and harvester images pin their own copies.
beautifulsoup4>=4.12
langdetect>=1.0.9
pyte>=0.8            # libs/claudeauth renders the CLI terminal; the oauth tests import it
pywebpush>=2.0       # services/notifier's browser notifications; the alert tests import it

# tests/test_e2e_site.py: headless Chromium walks the built site, served by uvicorn on a thread.
# **Capped**, because it decides pass or fail (CLAUDE.md §7): a Playwright release changes the
# browser protocol it speaks, and CHROME_BIN is a pinned build. Raise it with the walk run against
# the new version first.
playwright>=1.64,<1.65
uvicorn>=0.32
