# services/harvester runtime dependencies.
#
# Per-service, per CODE_STRUCTURE.md §5: this is the only image that will carry jobspy, just as
# docgen is the only one carrying Chrome. Keeping them separate is what lets api and notifier stay
# small and fast to rebuild, which matters because they are what gets restarted most.

httpx>=0.28
# Several adapters parse RSS/XML from third-party job boards. Stdlib ElementTree is documented as
# vulnerable to entity-expansion bombs, and the harvester shares a box with Postgres.
defusedxml>=0.7            # the fetch layer; per-host transport mounts are load-bearing for D3
pymongo>=4.19,<5
pydantic>=2.10         # libs/contracts config schema

# The JobSpy tier — LinkedIn, Indeed, Google Jobs. Listed as "added with the source adapters in
# step 4c, not before" and then not added, which is why those three sources could not run at all:
# the adapters were ported, the config existed, and the import would have failed on first call.
# Same pin as the reference.
python-jobspy>=1.1.80

# The description pass (`libs/corpus/postings.py`). It degrades to a regex tag-strip without this,
# which is not good enough to ship: the JSON-LD path is what gets the clean JD instead of the page's
# nav and cookie banner, and that text is what every label is formed from.
beautifulsoup4>=4.12
cryptography>=44.0       # libs/secrets, reached from libs/obs — every service that logs needs it

# The language filter (`libs/corpus/langfilter.py`). Pure Python, no models to download.
#
# **This image is the one that detects**, via `set_description` and `fill_missing_languages`, and it
# did not carry the library — the declaration stayed in the notifier's image after the code moved.
# The import failure was swallowed by the same blind `except` as an unreadable posting, so every
# row was stored with no language and every account's language filter admitted everything. Measured
# 2026-08-11: one account went from 608 detected in a day to 1.
langdetect>=1.0.9
